Port of ``docs/assets/make_figs_verified_v01.py``, ``make_figs_v03.py`` and ``make_figs_v04.py``. The plotting code is the
reviewed code; changed only: output format (300 dpi PNG plus PDF in ``figures/``), paths (``SACA_WORKSPACE``), and
the split into one function per figure. Every number is read from a frozen artifact (named in each figure) or is a
formula. Figures marked ILLUSTRATIVE use synthetic toy data and say so in the figure itself. Planning graphics
(GPU-budget projections) are not regenerated: they are superseded by the measured costs in the tables.

``python -m saca_rul.reporting.figures --out-dir figures`` writes every figure; ``--only NAME`` writes one.
"""

def load(rel):
    with open(os.path.join(WS, rel), encoding="utf-8") as f:
        return json.load(f)


def save(fig, name):
    """Write ``name``.png (300 dpi) and ``name``.pdf into the output folder."""
    os.makedirs(OUT, exist_ok=True)
    for ext in ("png", "pdf"):
        fig.savefig(os.path.join(OUT, f"{name}.{ext}"), dpi=300, bbox_inches="tight", facecolor="white")
    plt.close(fig)
    print("wrote", name)


def tci(x):
    x = np.asarray(x, float)
    m = x.mean()
    h = stats.t.ppf(0.975, len(x) - 1) * x.std(ddof=1) / np.sqrt(len(x))
    return m, m - h, m + h

def fig_r03_4_attention_illustrative() -> None:
    rng = np.random.default_rng(0)
    n = 40; d = 8
    Q = rng.normal(size=(n, d)); K = rng.normal(size=(n, d))
    S = Q @ K.T / np.sqrt(d)
    A_exact = np.exp(S - S.max(1, keepdims=True)); A_exact /= A_exact.sum(1, keepdims=True)
    u = min(5 * int(np.ceil(np.log(n))), n)  # 20 of 40 here; at theta* it is 30 of 150-240
    U = u
    idx = rng.integers(0, n, size=(n, U))
    M = np.array([S[i, idx[i]].max() - S[i, idx[i]].sum() / n for i in range(n)])
    top = np.argsort(-M)[:u]
    A_ps = np.full((n, n), 1.0 / n)
    A_ps[top] = A_exact[top]
    C = 13
    Qs = rng.normal(size=(C, d)); Ks = rng.normal(size=(C, d)); Ss = Qs @ Ks.T / np.sqrt(d)
    A_s = np.exp(Ss - Ss.max(1, keepdims=True)); A_s /= A_s.sum(1, keepdims=True)
    fig, ax = plt.subplots(1, 3, figsize=(12, 4))
    vmax = max(A_exact.max(), A_s.max())
    for a, M_, t in [(ax[0], A_ps, f"ProbSparse (θ*): {u} of {n} rows attend,\n{n-u} rows = uniform mean(V)"),
                     (ax[1], A_exact, f"Exact softmax (D1): all {n} rows attend"),
                     (ax[2], A_s, f"Sensor tokens (D2-D4): {C}×{C},\none token per sensor")]:
        im = a.imshow(M_, cmap="viridis", vmin=0, vmax=vmax)
        a.set_title(t, fontsize=9); a.set_xlabel("key token"); a.set_ylabel("query token")
    fig.colorbar(im, ax=ax, shrink=0.8, label="attention weight")
    fig.suptitle("ILLUSTRATIVE, SYNTHETIC: random queries/keys on a toy sequence; not a trained model", color="#a15c00", fontsize=10)
    save(fig, "fig_r03_4_attention_illustrative")

    vals = {s: [theta_best[s]] + [per[d][s]["best"] for d in ["D1", "D2", "D3", "D4"]] for s in subsets}
    sd = [float(np.mean([theta_best[s] for s in subsets]))] + [s1["S_D"][d] for d in ["D1", "D2", "D3", "D4"]]

    fig, ax = plt.subplots(1, 2, figsize=(12, 4.2))
    x = np.arange(len(designs))
    for s, m in zip(subsets, ["o", "s", "^"]):
        ax[0].plot(x, vals[s], marker=m, label=s)
    ax[0].plot(x, sd, "k--", lw=2, marker="D", label="mean over 3 subsets (S_D)")
    ax[0].set_xticks(x)
    ax[0].set_xticklabels(designs, rotation=20, ha="right", fontsize=9)
    ax[0].set_ylabel("best validation RMSE of 4 trials (cycles)")
    ax[0].set_title("(a) Best of study 0 per design (validation only)")
    ax[0].legend(fontsize=8)
    hy = s1["hypotheses"]
    labels = ["H1\nD1 − θ*", "H2\nD2 − D1", "H3\nD3 − D2", "H4\nD4 − D3"]
    med = [hy[h]["median_delta"] for h in ["H1", "H2", "H3", "H4"]]
    neg = [hy[h]["n_delta_negative"] for h in ["H1", "H2", "H3", "H4"]]
    cols = ["#b42318" if m > 0 else "#1d7a3a" for m in med]
    ax[1].bar(range(4), med, color=cols)
    for i, (m, n) in enumerate(zip(med, neg)):
        ax[1].text(i, m + (0.05 if m > 0 else -0.12), f"{m:+.2f}\n{n}/12 better", ha="center",
                   va="bottom" if m > 0 else "top", fontsize=8)
    ax[1].axhline(0, color="k", lw=0.8)
    ax[1].set_xticks(range(4))
    ax[1].set_xticklabels(labels, fontsize=9)
    ax[1].set_ylabel("median paired Δ validation RMSE")
    ax[1].set_ylim(-1.9, 1.1)
    ax[1].set_title("(b) Step-wise paired deltas, 12 non-independent pairs")
    fig.suptitle("Round 3 stage 1 (exp_B15): exploratory, validation only, FD001/FD003/FD004, one study per design", fontsize=10)
    save(fig, "fig_r04_1_chain")


def fig_r04_4_window_tokens() -> None:
    def tokens(L, P, S):
        n_code = (L - P) // S + 1  # P2 rule (theta* unfold, no padding)
        n_pad = (L - P) // S + 2   # D4 end padding
        return n_code, n_pad

    fig, axes = plt.subplots(2, 3, figsize=(12, 4.6), sharey=True)
    for r, (P, lab) in enumerate([(15, "FD001 (P = S = 15)"), (20, "FD002-FD004 (P = S = 20)")]):
        for c, L in enumerate([30, 40, 50]):
            ax = axes[r, c]
            n_code, n_pad = tokens(L, P, P)
            ax.add_patch(plt.Rectangle((0, 0.6), L, 0.3, color="#cfd8e3"))
            ax.add_patch(plt.Rectangle((L, 0.6), P, 0.3, color="#f3d9b1", hatch="//"))
            for n in range(n_pad):
                s = n * P
                ax.add_patch(plt.Rectangle((s, 0.1 + 0.0), P, 0.35, fill=False, ec="#0b6e99", lw=1.5))
                ax.text(s + P / 2, 0.27, f"p{n + 1}", ha="center", va="center", fontsize=7)
            ax.set_xlim(-1, 72)
            ax.set_ylim(0, 1.0)
            ax.set_yticks([])
            ok = "ok" if n_code >= 2 else "REJECTED"
            ax.set_title(f"{lab}, L={L}" + "\n" + f"P2 rule N_t={n_code} ({ok}); D4 N'={n_pad}", fontsize=7.5)
            if r == 1:
                ax.set_xlabel("cycle in window (grey = data; orange = end padding, last value repeated)", fontsize=6.5)
    fig.suptitle("D4 temporal patches with end padding (computed from N' = ⌊(L−P)/S⌋ + 2; frozen exp_B16 patch sizes)", fontsize=9)
    fig.tight_layout()
    save(fig, "fig_r04_4_window_tokens")


def fig_r04_5_bn_ln_illustrative() -> None:
    rng = np.random.default_rng(0)
    B, S, d = 6, 5, 8
    Z = rng.normal(0, 1, (B, S, d)) + np.linspace(-2, 2, B)[:, None, None]
    bn = (Z - Z.mean(axis=(0, 1), keepdims=True)) / Z.std(axis=(0, 1), keepdims=True)
    ln = (Z - Z.mean(axis=2, keepdims=True)) / Z.std(axis=2, keepdims=True)
    fig, ax = plt.subplots(1, 3, figsize=(11, 3.2))
    for a, M, t in zip(ax, [Z, bn, ln], ["input Z (token means shifted per sample)",
                                        "BatchNorm: per feature j over all (b, s)",
                                        "LayerNorm: per token (b, s) over d features"]):
        im = a.imshow(M.reshape(B * S, d), aspect="auto", cmap="coolwarm", vmin=-3, vmax=3)
        a.set_title(t, fontsize=8)
        a.set_xlabel("feature j")
        a.set_ylabel("token (b, s)")
    fig.colorbar(im, ax=ax, shrink=0.8)
    fig.suptitle("ILLUSTRATIVE, SYNTHETIC: which statistics each norm uses (random numbers, not a trained model)", fontsize=9, color="#a15c00")
    save(fig, "fig_r04_5_bn_ln_illustrative")


def fig_score_function() -> None:
    """NASA Score per engine as a function of d = predicted - true RUL: the corrected constants (early /13, late /10)
    against the swapped constants of the submitted code (early /10, late /13). Computed from the formula (R0.6)."""
    from ..evaluation import A_EARLY, A_LATE

    d = np.linspace(-50, 50, 401)
    ok = np.where(d < 0, np.expm1(-d / A_EARLY), np.expm1(d / A_LATE))
    swapped = np.where(d < 0, np.expm1(-d / 10.0), np.expm1(d / 13.0))
    fig, ax = plt.subplots(figsize=(6.5, 4.2))
    ax.plot(d, ok, color="#0b6e99", lw=2, label="this work: early /13, late /10")
    ax.plot(d, swapped, color="#b42318", lw=1.6, ls="--", label="submitted code: early /10, late /13 (swapped)")
    ax.axvline(0, color="k", lw=0.6)
    ax.set_xlabel("d = predicted RUL - true RUL (cycles); d < 0 early, d >= 0 late")
    ax.set_ylabel("score contribution of one engine")
    ax.set_ylim(0, 160)
    ax.set_title("NASA Score: computed from the formula (not a result)", fontsize=10)
    ax.legend(fontsize=8)
    ax.grid(alpha=0.3)
    save(fig, "fig_score_function")
