"""Training under protocol P2: one data path, one training loop, one set of artifacts for every model.

``train_one`` is the full-fidelity loop (MSE, AdamW, gradient clip 1.0, no scheduler, no input noise). Early stopping and
the returned value use RMSE over the common-support validation windows (label cycle >= 51; Algorithm 1 steps 3-4, 6).
``run_training`` is the outer fit (Algorithm 1 step 6): retrain the frozen configuration of one subset on split seed
``seed`` and write ``ckpt_rmse.pt``, ``history.csv``, ``preds_val.csv``, ``train.log`` and ``train_meta.json``.

Strict determinism is on for every run (``determinism.set_all``). The test files are never opened here: ``run_training``
installs the test guard before it touches any data.
"""
from __future__ import annotations

import datetime as dt
import hashlib
import json
import math
import platform
import time
from collections.abc import Callable
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import yaml

from . import baselines as B
from . import checkpoint as CK
from . import determinism, guard
from . import evaluation as EV
from . import model as M
from .data import DataCache
from .embeddings import sensor_ids_from_features
from .search_space import Infeasible, InfeasibleConfig, is_r3, materialize, space_family

EVAL_BS_DEFAULT = 256


class TrialTimeout(RuntimeError):
    """Wall-clock deadline of a tuning study was hit."""


def sha256_file(p: str | Path) -> str:
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for c in iter(lambda: f.read(1 << 20), b""):
            h.update(c)
    return h.hexdigest()


def load_protocol(path: str | Path) -> dict[str, Any]:
    """Load a protocol YAML; adds ``_path``, ``_sha256`` and ``_eval_bs``."""
    with open(path, encoding="utf-8") as f:
        pr: dict[str, Any] = yaml.safe_load(f)
    pr["_path"] = str(path)
    pr["_sha256"] = sha256_file(path)
    pr["_eval_bs"] = int(pr.get("eval", {}).get("batch_size", EVAL_BS_DEFAULT))
    return pr


def n_params(model: nn.Module) -> int:
    return int(sum(p.numel() for p in model.parameters()))


def config_hash(obj: Any) -> str:
    return hashlib.sha256(json.dumps(obj, sort_keys=True, default=str).encode()).hexdigest()


# ------------------------------------------------------------------------------------------------ build
def build_for_family(family: str, cfg: dict[str, Any], C: int, L: int, sensor_ids: tuple[int, ...] | None = None,
                     norm: str = "batch") -> tuple[nn.Module, dict[str, Any] | M.ModelConfig]:
    """Build the model of a P2 family. Returns ``(model, mc)`` where ``mc`` is the checkpoint config."""
    if is_r3(family) or family == "dualdim":
        try:
            mc = M.resolve(dict(cfg), C=C, L=L)
            design = "theta" if family == "dualdim" else family[3:]
            return M.build_model(mc, design, sensor_ids, norm=norm), mc
        except ValueError as e:
            if type(e).__name__ == "InfeasibleConfig" or "patch" in str(e).lower():
                raise Infeasible(str(e)) from e
            raise
    try:
        return B.build_baseline(family, dict(cfg), C, L), dict(cfg, family=family, C=C, L=L)
    except InfeasibleConfig as e:
        raise Infeasible(str(e)) from e


def sensor_ids_for(family: str, features: list[str]) -> tuple[int, ...] | None:
    return sensor_ids_from_features(features) if is_r3(family) else None


# ------------------------------------------------------------------------------------------------ OOM rule
def is_oom(e: BaseException) -> bool:
    """True for CUDA out-of-memory errors (torch OOM class or allocator/cuBLAS messages)."""
    if isinstance(e, torch.cuda.OutOfMemoryError):
        return True
    s = str(e)
    return isinstance(e, RuntimeError) and ("out of memory" in s or "CUBLAS_STATUS_ALLOC_FAILED" in s)


def apply_memory_cap(device: torch.device, protocol: dict[str, Any]) -> float | None:
    """Cap the caching allocator at a fixed fraction of device memory (same cap for every model), so a configuration
    that does not fit raises OOM instead of spilling into shared system memory. Returns the cap in MB."""
    if device.type != "cuda":
        return None
    frac = float(protocol.get("oom_rule", {}).get("memory_fraction", 0.90))
    idx = device.index if device.index is not None else torch.cuda.current_device()
    torch.cuda.set_per_process_memory_fraction(frac, idx)
    return frac * torch.cuda.get_device_properties(idx).total_memory / 2 ** 20


def predict_safe(model: nn.Module, X: np.ndarray, bs: int, device: torch.device | None = None,
                 log: Callable[[str], None] | None = None) -> np.ndarray:
    """``checkpoint.predict`` at batch ``bs``; on CUDA OOM halve the batch and retry (predictions are batch-invariant)."""
    bs = int(bs)
    while True:
        try:
            return CK.predict(model, X, bs, device=device)
        except Exception as e:  # noqa: BLE001
            if not is_oom(e) or bs <= 1:
                raise
            torch.cuda.empty_cache()
            bs //= 2
            if log:
                log(f"eval OOM: eval batch halved to {bs}")


# ------------------------------------------------------------------------------------------------ loop
def train_one(model: nn.Module, w: dict[str, Any], recipe: dict[str, Any], device: torch.device, seed: int,
              max_epochs: int, patience: int, eval_bs: int = EVAL_BS_DEFAULT, deadline: float | None = None,
              log: Callable[[str], None] | None = None,
              on_best: Callable[[int, nn.Module], None] | None = None) -> tuple[float, int, list[dict[str, Any]]]:
    """Returns ``(best common-support validation RMSE, best epoch, history)``."""
    model.to(device)
    Xtr = torch.from_numpy(w["Xtr"]).to(device)
    ytr = torch.from_numpy(w["ytr"]).to(device)
    sup = w["support"]
    crit = nn.MSELoss()
    opt = torch.optim.AdamW(model.parameters(), lr=float(recipe["lr"]), weight_decay=float(recipe["weight_decay"]))
    bs = int(recipe["batch_size"])
    gen = torch.Generator(device=device)
    gen.manual_seed(int(seed))
    N = Xtr.shape[0]
    best, best_ep, no_imp = float(np.inf), 0, 0
    hist: list[dict[str, Any]] = []
    if device.type == "cuda":
        torch.cuda.reset_peak_memory_stats()
    for ep in range(1, int(max_epochs) + 1):
        t0 = time.time()
        model.train()
        perm = torch.randperm(N, device=device, generator=gen)
        tot = torch.zeros((), device=device)
        for i in range(0, N, bs):
            idx = perm[i:i + bs]
            opt.zero_grad(set_to_none=True)
            loss = crit(model(Xtr[idx]).reshape(-1), ytr[idx])
            loss.backward()
            nn.utils.clip_grad_norm_(model.parameters(), 1.0)
            opt.step()
            tot += loss.detach() * len(idx)
        pv = predict_safe(model, w["Xva"], eval_bs, device=device, log=log)
        v_sup = EV.rmse(w["yva"][sup], pv[sup])
        v_all = EV.rmse(w["yva"], pv)
        rec = {"epoch": ep, "train_loss": float(tot) / N, "val_RMSE_support": v_sup, "val_RMSE_allwin": v_all,
               "epoch_s": time.time() - t0, "steps": int(math.ceil(N / bs)),
               "peak_mem_MB": (torch.cuda.max_memory_allocated() / 2 ** 20) if device.type == "cuda" else 0.0}
        hist.append(rec)
        improved = v_sup < best - 1e-6
        if improved:
            best, best_ep, no_imp = v_sup, ep, 0
            if on_best is not None:
                on_best(ep, model)
        else:
            no_imp += 1
        if log:
            log(f"ep {ep:3d} loss {rec['train_loss']:.3f} valRMSE_sup {v_sup:.4f} all {v_all:.4f} "
                f"{'*' if improved else ' '} {rec['epoch_s']:.1f}s")
        if no_imp >= int(patience):
            break
        if deadline is not None and time.time() > deadline:
            raise TrialTimeout(f"wall-clock deadline hit at epoch {ep} (best {best:.4f} at {best_ep})")
    return best, best_ep, hist


# ------------------------------------------------------------------------------------------------ outer fit
@dataclass
class RunSpec:
    """Everything ``run_training`` needs. ``arm`` carries the one-factor changes of the ablation arms."""

    experiment_id: str
    family: str
    subset: str
    seed: int
    params: dict[str, Any]
    protocol_path: Path
    data_dir: Path
    splits_path: Path
    out_dir: Path
    device: str = "cuda"
    max_epochs: int | None = None  # smoke only; capped runs are never frozen
    selection_sha256: str = ""
    trial_id: str | None = None
    arm: dict[str, Any] = field(default_factory=dict)


def run_training(spec: RunSpec) -> dict[str, Any]:
    """Retrain one (subset, seed) and write the run folder. Returns the ``train_meta`` dict."""
    guard.install()  # the test split cannot be opened from here on
    pr = load_protocol(spec.protocol_path)
    arm = spec.arm
    smoke = spec.max_epochs is not None
    tag = f"{spec.subset}_seed{spec.seed}" + (f"_order{spec.seed}" if arm.get("sensor_order") else "") \
        + ("_smoke" if smoke else "")
    run_dir = Path(spec.out_dir) / "runs" / tag
    run_dir.mkdir(parents=True, exist_ok=True)
    logf = open(run_dir / "train.log", "a", encoding="utf-8")

    def log(msg: str) -> None:
        line = f"[{dt.datetime.now().strftime('%H:%M:%S')}] {msg}"
        print(line, flush=True)
        logf.write(line + "\n")
        logf.flush()

    seed_info = determinism.set_all(spec.seed)
    device = torch.device(spec.device if (spec.device == "cpu" or torch.cuda.is_available()) else "cpu")
    mem_cap_mb = apply_memory_cap(device, pr)
    train_engines = None
    if arm.get("tr_subsample"):
        sz = json.loads(Path(arm["tr_subsample"]["file"]).read_text(encoding="utf-8"))
        train_engines = [int(e) for e in sz["subsets"][spec.subset][str(spec.seed)][str(arm["tr_subsample"]["fraction"])]]
    cache = DataCache(spec.subset, spec.seed, spec.data_dir, spec.splits_path, smoothing=pr["data"]["smoothing"],
                      alpha=float(pr["data"]["ema_alpha"]),
                      support_min_label_cycle=int(pr["objective"]["support_min_label_cycle"]),
                      train_engines=train_engines)
    order_ent: dict[str, Any] | None = None
    if arm.get("sensor_order"):
        od = json.loads(Path(arm["sensor_order"]).read_text(encoding="utf-8"))
        order_ent = od["orders"][spec.subset][str(spec.seed)]
        if int(order_ent["split_seed"]) != spec.seed:
            raise ValueError(f"order {spec.seed} is bound to split seed {order_ent['split_seed']}")
        cache.apply_sensor_order(order_ent["perm"], order_ent["original_features"], order_ent["permuted_features"])
        log(f"sensor order {spec.seed} applied: {cache.features}")
    bypass = arm.get("bypass_buildability")
    cfg, L, rec = materialize(spec.family, spec.params, cache.C, pr["fixed"].get(space_family(spec.family)),
                              bypass_buildability=bool(bypass), bypass_L=arm.get("bypass_L"))
    sids = sensor_ids_for(spec.family, cache.features)
    model, mc = build_for_family(spec.family, cfg, cache.C, L, sids, norm=arm.get("norm", "batch"))
    recipe = dict(pr["recipe"], **rec)
    if spec.max_epochs:
        recipe["max_epochs"] = spec.max_epochs
    w = cache.windows(L)
    chash = config_hash({"family": spec.family, "cfg": cfg, "recipe": recipe, "C": cache.C})
    import sklearn
    code_hashes = M.code_sha256() if (is_r3(spec.family) or spec.family == "dualdim") else B.code_sha256()
    hdr = {"experiment": spec.experiment_id, "subset": spec.subset, "seed": spec.seed, "smoke": smoke,
           "family": spec.family, "selection_sha256": spec.selection_sha256, "selected_trial": spec.trial_id,
           "params": json.dumps(spec.params, sort_keys=True), "config_hash": chash,
           "recipe": json.dumps(recipe, sort_keys=True), "protocol_sha256": pr["_sha256"],
           "features": cache.features, "C": cache.C, "L": L, "n_train_windows": len(w["ytr"]),
           "n_val_windows": len(w["yva"]), "n_val_support": int(w["support"].sum()), "n_params": n_params(model),
           "seeding": seed_info, "code_sha256": json.dumps(code_hashes, sort_keys=True),
           "sensor_ids": list(sids) if sids is not None else None, "arm": json.dumps(arm, sort_keys=True, default=str),
           "mem_cap_MB": mem_cap_mb, "eval_batch": pr["_eval_bs"]}
    logf.write(f"# run started {dt.datetime.now().isoformat(timespec='seconds')}\n# python {platform.python_version()} "
               f"torch {torch.__version__} numpy {np.__version__} pandas {pd.__version__} sklearn {sklearn.__version__}\n"
               f"# cuda {torch.version.cuda} device {torch.cuda.get_device_name(0) if device.type == 'cuda' else 'cpu'}\n"
               f"# data_splits.json sha256 {sha256_file(spec.splits_path)}\n# git: none\n"
               + "".join(f"# {k}: {v}\n" for k, v in hdr.items()))
    ck = run_dir / "ckpt_rmse.pt"
    extra: dict[str, Any] = {"features": cache.features, "L": L, "C": cache.C, "subset": spec.subset,
                             "seed": spec.seed,
                             "preprocess": {"smoothing": pr["data"]["smoothing"], "alpha": float(pr["data"]["ema_alpha"]),
                                            "legacy_corr": False},
                             "experiment_id": spec.experiment_id, "config_hash": chash, "recipe": recipe,
                             "family": spec.family, "data_splits_sha256": sha256_file(spec.splits_path)}
    if order_ent is not None:
        extra["sensor_order"] = order_ent
    if train_engines is not None:
        extra["tr_engines_override"] = train_engines

    def on_best(ep: int, m: nn.Module) -> None:
        CK.save_ckpt(ck, spec.family, m, mc, recipe, dict(extra, best_epoch=ep, selected_by="val_rmse_support"))

    t0 = time.time()
    best, best_ep, hist = train_one(model, w, recipe, device, spec.seed, recipe["max_epochs"], recipe["patience"],
                                    eval_bs=pr["_eval_bs"], log=log, on_best=on_best)
    ttime = time.time() - t0
    pd.DataFrame(hist).to_csv(run_dir / "history.csv", index=False)
    mdl, _ = CK.load_for_eval(ck, device=str(device), norm=arm.get("norm", "batch"))
    pv = predict_safe(mdl, w["Xva"], pr["_eval_bs"], device=device, log=log)
    last = pd.Series(w["cva"]).groupby(w["eva"]).transform("max").to_numpy() == w["cva"]
    pd.DataFrame({"engine_id": w["eva"], "cycle_label": w["cva"], "y_true_capped": w["yva"], "y_pred": pv,
                  "last_window": last, "on_support": w["support"]}).to_csv(
        run_dir / "preds_val.csv", index=False, float_format="%.6f")
    st = pd.read_csv(run_dir / "preds_val.csv")
    sup = st.on_support.to_numpy()
    vm = EV.all_metrics(st.y_true_capped.to_numpy()[sup], st.y_pred.to_numpy()[sup])
    vm_all = EV.all_metrics(st.y_true_capped.to_numpy(), st.y_pred.to_numpy())
    meta = dict(extra, best_epoch_rmse=best_ep, epochs_run=len(hist), train_time_s=ttime,
                s_per_epoch=float(np.mean([h["epoch_s"] for h in hist])),
                peak_mem_MB=float(max(h["peak_mem_MB"] for h in hist)), n_params=n_params(model),
                n_train_windows=int(len(w["ytr"])), n_val_windows=int(len(w["yva"])), val=vm, val_allwin=vm_all,
                val_metric="common support (label cycle >= 51)", ckpt_rmse_sha256=sha256_file(ck),
                smoke=smoke, finished=dt.datetime.now().isoformat(timespec="seconds"),
                device=torch.cuda.get_device_name(0) if device.type == "cuda" else "cpu",
                test_guard_blocked=guard.blocked_count())
    (run_dir / "train_meta.json").write_text(json.dumps(meta, indent=1, default=str), encoding="utf-8")
    log(f"done: {len(hist)} epochs, {ttime / 60:.1f} min, best support val RMSE {vm['RMSE']:.4f} (ep {best_ep}); "
        f"recomputed from file {vm['RMSE']:.6f} vs loop {best:.6f}")
    logf.close()
    return meta


def state_dict_sha256(path: str | Path) -> str:
    """sha256 over the tensors of a checkpoint's state dict (name, dtype, shape, raw bytes; names sorted, as in the R4-CERT comparison), independent of file layout."""
    obj = torch.load(path, map_location="cpu", weights_only=False)
    h = hashlib.sha256()
    for k in sorted(obj["state_dict"]):
        t = obj["state_dict"][k].detach().cpu().contiguous()
        h.update(k.encode())
        h.update(str(t.dtype).encode())
        h.update(str(tuple(t.shape)).encode())
        h.update(t.numpy().tobytes())
    return h.hexdigest()


def run_ridge(spec: RunSpec) -> dict[str, Any]:
    """Ridge refit (Algorithm 1 step 6 for ridge): the frozen ``(L, alpha)`` fitted on TR of split seed ``seed``.

    CPU, float64, closed form (no random state), so the run is bitwise repeatable. Writes the same run folder as
    ``run_training`` (no ``history.csv``)."""
    from sklearn.linear_model import Ridge

    guard.install()
    pr = load_protocol(spec.protocol_path)
    run_dir = Path(spec.out_dir) / "runs" / f"{spec.subset}_seed{spec.seed}"
    run_dir.mkdir(parents=True, exist_ok=True)
    L, alpha = int(spec.params["L"]), float(spec.params["alpha"])
    cache = DataCache(spec.subset, spec.seed, spec.data_dir, spec.splits_path, smoothing=pr["data"]["smoothing"],
                      alpha=float(pr["data"]["ema_alpha"]),
                      support_min_label_cycle=int(pr["objective"]["support_min_label_cycle"]))
    w = cache.windows(L)
    flat = lambda X: np.ascontiguousarray(X.reshape(len(X), -1), dtype=np.float64)  # noqa: E731
    est = Ridge(alpha=alpha, solver="cholesky", fit_intercept=True).fit(flat(w["Xtr"]), w["ytr"].astype(np.float64))
    m = B.ridge_from_sklearn(est, cache.C, L)
    recipe = {"estimator": "sklearn.linear_model.Ridge", "alpha": alpha, "solver": "cholesky", "fit_intercept": True,
              "dtype": "float64", "L": L}
    chash = config_hash({"family": "ridge", "cfg": {"alpha": alpha}, "recipe": recipe, "C": cache.C})
    extra = {"features": cache.features, "L": L, "C": cache.C, "subset": spec.subset, "seed": spec.seed,
             "preprocess": {"smoothing": pr["data"]["smoothing"], "alpha": float(pr["data"]["ema_alpha"]),
                            "legacy_corr": False},
             "experiment_id": spec.experiment_id, "config_hash": chash, "recipe": recipe, "family": "ridge",
             "data_splits_sha256": sha256_file(spec.splits_path)}
    ck = run_dir / "ckpt_rmse.pt"
    CK.save_ckpt(ck, "ridge", m, {"alpha": alpha, "family": "ridge", "C": cache.C, "L": L}, recipe, extra)
    mm, _ = CK.load_for_eval(ck)
    p_main = CK.predict(mm, w["Xva"], 1024)
    ref = est.predict(flat(w["Xva"]))
    loader_diff = float(np.abs(p_main - ref).max())
    last = pd.Series(w["cva"]).groupby(w["eva"]).transform("max").to_numpy() == w["cva"]
    pd.DataFrame({"engine_id": w["eva"], "cycle_label": w["cva"], "y_true_capped": w["yva"], "y_pred": p_main,
                  "last_window": last, "on_support": w["support"]}).to_csv(
        run_dir / "preds_val.csv", index=False, float_format="%.6f")
    st = pd.read_csv(run_dir / "preds_val.csv")
    sup = st.on_support.to_numpy()
    vm = EV.all_metrics(st.y_true_capped.to_numpy()[sup], st.y_pred.to_numpy()[sup])
    meta = dict(extra, best_epoch_rmse=None, epochs_run=0, n_params=cache.C * L + 1, val=vm,
                val_metric="common support (label cycle >= 51)", ckpt_rmse_sha256=sha256_file(ck), smoke=False,
                device="cpu", loader_check_maxdiff_vs_sklearn=loader_diff)
    (run_dir / "train_meta.json").write_text(json.dumps(meta, indent=1, default=str), encoding="utf-8")
    return meta
